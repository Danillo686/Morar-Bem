Banco (pgAdmin):
CREATE TABLE imoveis (
	codigo VARCHAR (50) PRIMARY KEY,
	titulo VARCHAR (150) NOT NULL, 
	preco numeric(10, 2) NOT NULL
);

CREATE TABLE leads (
	id SERIAL PRIMARY KEY,
	id_imovel VARCHAR (50) NOT NULL,
	telefone VARCHAR (20) NOT NULL,
	data_descricao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	CONSTRAINT fk_imovel FOREIGN KEY (id_imovel) REFERENCES imoveis(codigo) ON DELETE CASCADE 
	--Ela diz para o banco: 
	--"O valor inserido em id_imovel obrigatoriamente precisa existir na coluna codigo da tabela imoveis".
	--O ON DELETE CASCADE significa que, se a imobiliária deletar um imóvel do sistema, todos
	--os contatos daquele imóvel são apagados juntos de forma automática para não deixar dados órfãos.	
);
